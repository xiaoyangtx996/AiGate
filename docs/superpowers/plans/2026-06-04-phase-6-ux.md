<script setup lang="ts">
const { search } = useAigateApi()
const searchQuery = ref('')
const results = ref<any[]>([])
const isOpen = ref(false)
const loading = ref(false)

// 防抖
let timeout: NodeJS.Timeout
watch(searchQuery, (newQuery) => {
  clearTimeout(timeout)
  if (newQuery.length < 2) {
    results.value = []
    return
  }
  
  timeout = setTimeout(async () => {
    loading.value = true
    try {
      const res = await search(newQuery)
      results.value = res.data || []
    }
    finally {
      loading.value = false
    }
  }, 300)
})

// 快捷键 Cmd/Ctrl + K
onMounted(() => {
  window.addEventListener('keydown', (e) => {
    if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
      e.preventDefault()
      isOpen.value = true
    }
  })
})

function close() {
  isOpen.value = false
  searchQuery.value = ''
  results.value = []
}
</script>

<template>
  <UModal v-model:open="isOpen">
    <UCommandPalette
      v-model="searchQuery"
      :loading
      :groups="[
        {
          id: 'results',
          label: '搜索结果',
          items: results.map(r => ({
            id: r.id,
            label: r.name,
            description: r.type,
            onSelect: () => navigateTo(r.url),
          })),
        },
      ]"
      placeholder="搜索 Agents、Channels、API Keys..."
      @close="close"
    />
  </UModal>
</template>

```typescript
export default defineEventHandler(async (event) => {
  const query = getQuery(event).q as string
  if (!query || query.length < 2) {
    return responseSuccess([])
  }
  
  const results = await Promise.all([
    // 搜索 Agents
    db.select().from(agent).where(ilike(agent.name, `%${query}%`)).limit(5),
    // 搜索 Channels
    db.select().from(channel).where(ilike(channel.name, `%${query}%`)).limit(5),
    // 搜索 API Keys
    db.select().from(apiKey).where(ilike(apiKey.name, `%${query}%`)).limit(5),
  ])
  
  const formatted = [
    ...results[0].map((a: any) => ({ ...a, type: 'Agent', url: `/aigate/agents/${a.id}` })),
    ...results[1].map((c: any) => ({ ...c, type: 'Channel', url: `/aigate/channels/${c.id}` })),
    ...results[2].map((k: any) => ({ ...k, type: 'API Key', url: `/aigate/api-keys/${k.id}` })),
  ]
  
  return responseSuccess(formatted)
})
```

创建 `AiGate-app/app/composables/useBatchOperations.ts`:
```typescript
export function useBatchOperations<T>(options: {
  onDelete?: (items: T[]) => Promise<void>
  onExport?: (items: T[]) => void
  onStatusChange?: (items: T[], status: string) => Promise<void>
}) {
  const selectedIds = ref<Set<string>>(new Set())
  const isAllSelected = computed({
    get: () => selectedIds.value.size > 0,
    set: (value: boolean) => {
      if (!value) selectedIds.value.clear()
    },
  })
  
  function toggleSelect(id: string) {
    if (selectedIds.value.has(id)) {
      selectedIds.value.delete(id)
    }
    else {
      selectedIds.value.add(id)
    }
  }
  
  function toggleSelectAll(ids: string[]) {
    if (selectedIds.value.size === ids.length) {
      selectedIds.value.clear()
    }
    else {
      selectedIds.value = new Set(ids)
    }
  }
  
  async function batchDelete(items: T[]) {
    if (!confirm(`Delete ${items.length} items?`)) return
    await options.onDelete?.(items)
    selectedIds.value.clear()
  }
  
  async function batchStatusChange(items: T[], status: string) {
    await options.onStatusChange?.(items, status)
    selectedIds.value.clear()
  }
  
  return {
    selectedIds,
    isAllSelected,
    toggleSelect,
    toggleSelectAll,
    batchDelete,
    batchStatusChange,
  }
}
```

```typescript
import * as XLSX from 'xlsx'

export function useExport() {
  function exportToCSV<T>(data: T[], filename: string) {
    // 展平嵌套对象
    const flattened = data.map(item => {
      const flat: Record<string, unknown> = {}
      for (const [key, value] of Object.entries(item)) {
        if (typeof value === 'object' && value !== null) {
          Object.assign(flat, { [`${key}.${k}`]: v } for k in Object.entries(value))
        }
        else {
          flat[key] = value
        }
      }
      return flat
    })
    
    const worksheet = XLSX.utils.json_to_sheet(flattened)
    const workbook = XLSX.utils.book_new()
    XLSX.utils.book_append_sheet(workbook, worksheet, 'Sheet1')
    XLSX.writeFile(workbook, `${filename}.csv`)
  }
  
  function exportToExcel<T>(data: T[], filename: string) {
    const worksheet = XLSX.utils.json_to_sheet(data)
    const workbook = XLSX.utils.book_new()
    XLSX.utils.book_append_sheet(workbook, worksheet, 'Sheet1')
    XLSX.writeFile(workbook, `${filename}.xlsx`)
  }
  
  function exportToJSON<T>(data: T[], filename: string) {
    const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' })
    const url = URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    a.download = `${filename}.json`
    a.click()
    URL.revokeObjectURL(url)
  }
  
  return {
    exportToCSV,
    exportToExcel,
    exportToJSON,
  }
}
```
